/* Q:- How to find employees who earn more 
than the average salary of their department? */

-- Creating Employees__ table
CREATE TABLE Employees__
(EmployeeID INT,EmployeeName VARCHAR(20),Salary INT,DepartmentID int)

-- Inserting values into Employees__ table
INSERT INTO Employees__ VALUES
(1,'Pavithra',20000,1),
(2,'Rahul',12000,1),
(3,'Pallavi',50000,2),
(4,'Chandra',45000,2),
(5,'Priyanka',30000,2),
(6,'Dhanu',15000,3),
(7,'Chinnu',15500,3);

-- Ans:-
SELECT * FROM (
	SELECT
		EmployeeID,
		EmployeeName,
		Salary,
		DepartmentID,
		AVG(Salary) OVER(PARTITION BY DepartmentID) AS AverageSalary
	FROM Employees__ )t
WHERE Salary > AverageSalary;
