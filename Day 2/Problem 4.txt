-- Creating CustomerStatus Table
Create Table CustomerStatus
(CustomerID INT,Status VARCHAR(8),Date Date)

-- Ans:-
WITH CustomerLastDate AS (
SELECT
CustomerID,
MAX(Date) AS LastDate
FROM CustomerStatus
GROUP BY CustomerID )

SELECT
C.CustomerID
FROM CustomerStatus AS C
JOIN CustomerLastDate AS CL
ON C.CustomerID = CL.CustomerID
WHERE 
C.Date = CL.LastDate AND
C.Status = 'Active'
